# ブランチ運用時のマージ手順

``` powershell
# ==================================================
# 【工程0】作業開始前チェック
# ==================================================
git status
# 作業中の変更が残っていないか確認
# 残っていたら commit または stash する

# ==================================================
# 【工程1】最新の main から個人ブランチを作成
# ==================================================
git checkout main
# 共有ブランチ main に切り替える

git fetch origin
# リモートリポジトリの最新状態を取得（まだ反映しない）

git pull --rebase origin main
# リモート main を取り込み、ローカル main を最新化
# 履歴を一直線に保つ

git checkout -b feature/xxx
# 最新の main から個人作業ブランチを新規作成

# ==================================================
# 【工程2】作業・コミット
# ==================================================
git add <file>
# 変更したファイルをステージング

git commit -m "feat: xxx を追加"
# 作業内容を小さく区切ってコミットする

# ==================================================
# 【工程3】作業中の定期同期（競合の早期解消）
# ==================================================
git fetch origin
# 他メンバーの更新を取得

git rebase origin/main
# 個人ブランチに最新 main を取り込む
# 競合はここで解消する

# （競合が発生した場合）
git status
# 競合しているファイルを確認

# エディタで競合を解消後
git add <file>
# 解消したファイルをステージング

git rebase --continue
# rebase を再開

# ==================================================
# 【工程4】作業完了前の最終同期（マージ前チェック）
# ==================================================
git fetch origin
# マージ直前にもう一度最新状態を取得

git rebase origin/main
# main の最新変更をすべて取り込み
# マージ時の競合を防止する

# ==================================================
# 【工程5】個人ブランチをリモートへ push
# ==================================================
git push origin feature/xxx
# 初回 push

# （rebase 後に push 済みだった場合）
git push --force-with-lease
# 安全に履歴を書き換えて push

# ==================================================
# 【工程6】main へマージ（共有ブランチ）
# ==================================================
git checkout main
# main に切り替える

git pull origin main
# main を最新状態にする

git merge --no-ff feature/xxx
# 個人ブランチを main にマージ
# マージコミットを必ず残す

git push origin main
# マージ結果をリモートに反映

# ==================================================
# 【工程7】後片付け（ブランチ削除）
# ==================================================
git branch -d feature/xxx
# ローカルの作業ブランチを削除

git push origin --delete feature/xxx
# リモートの作業ブランチも削除
```
